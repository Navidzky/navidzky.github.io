% ======================= JOB MARKET PAPER =======================
\section{Job Market Paper}

\paper{The Role of the Remote Matching in Transport Markets}
\begin{desc}
  \item Studies how carriers and customers search for and match in transportation markets, where
  carriers are differentiated by location and customers by origin--destination pair. Develops a
  dynamic spatial directed search model showing how market conditions determine whether customers
  or carriers bear the cost of the pickup trip, and how remote matching links conditions across
  locations through the carriers markets share.
\end{desc}

% ======================= WORKING PAPERS =======================
\section{Working Papers}

\paper{\href{https://dx.doi.org/10.2139/ssrn.6379739}{When Outside Options Shape the Labor Market: A Directed Search Perspective}}
\begin{desc}
  \item Directed search model with skilled/unskilled workers and high-/low-tech firms shows how
  skilled workers' outside options raise posted wages but can reduce high-tech firm entry,
  spilling over to unskilled workers through firm composition and wage competition.
\end{desc}

\paper{\href{https://dx.doi.org/10.2139/ssrn.4830956}{The Heterogeneous Effects of Exogenous Shocks on Public Transportation Demand}}
\begin{desc}
  \item With Kowsar Yousefi and Amir Samimi. A gasoline price shock raised subway usage by
  roughly 3.9\%; usage fell by roughly 42.6\% during the COVID-19 pandemic.
\end{desc}
