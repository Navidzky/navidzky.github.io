% ======================= JOB MARKET PAPER =======================
\section{Job Market Paper}

\paper{The Economics of Chokepoints: Remote Matching and Entry Deterrence in Global Oil Tanker Markets}
\begin{desc}
  \item Maritime chokepoints act as an entry cost, not only a transport cost: because tankers are
  chartered remotely, days in advance, carriers commit to distant markets under uncertainty, so an
  anticipated closure deters entry before any physical disruption occurs. Distinguishes gateways
  (irreplaceable, e.g.\ Strait of Hormuz) from passages (substitutable, e.g.\ Bab el-Mandeb); the
  2026 Strait of Hormuz restriction supplies the identifying episode.
\end{desc}

% ======================= WORKING PAPERS =======================
\section{Working Papers}

\paper{When Outside Options Shape the Labor Market: A Directed Search Perspective}
\begin{desc}
  \item Directed search model with skilled/unskilled workers and high-/low-tech firms shows how
  skilled workers' outside options raise posted wages but can reduce high-tech firm entry,
  spilling over to unskilled workers through firm composition and wage competition.
\end{desc}

\paper{\href{https://dx.doi.org/10.2139/ssrn.4830956}{The Heterogeneous Effects of Exogenous Shocks on Public Transportation Demand}}
\begin{desc}
  \item With Kowsar Yousefi and Amir Samimi. A gasoline price shock raised subway usage by
  roughly 3.9\%; usage fell by roughly 42.6\% during the COVID-19 pandemic.
\end{desc}
